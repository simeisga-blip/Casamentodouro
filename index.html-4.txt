                    localStorage.setItem('w_config', JSON.stringify(this.config));
                    localStorage.setItem('w_budget', JSON.stringify(this.budget));
                    localStorage.setItem('w_suppliers', JSON.stringify(this.suppliers));
                    localStorage.setItem('w_pagamentos', JSON.stringify(this.pagamentos));
                    localStorage.setItem('w_quintas', JSON.stringify(this.quintas));
                    localStorage.setItem('w_guests', JSON.stringify(this.guests));
                    localStorage.setItem('w_mesas', JSON.stringify(this.mesas));
                    localStorage.setItem('w_tasks', JSON.stringify(this.tasks));
                    localStorage.setItem('w_honeymoon', JSON.stringify(this.honeymoon));
                    localStorage.setItem('w_gifts', JSON.stringify(this.gifts));
                    localStorage.setItem('w_inspirations', JSON.stringify(this.inspirations));
                },

                handleLogin() {
                    if(!this.emailInput || !this.emailInput.includes('@')) {
                        this.loginError = true;
                        this.errorMessage = 'Por favor, insira um e-mail válido.';
                        return;
                    }
                    localStorage.setItem('wedding_auth', 'true');
                    this.isLoggedIn = true;
                    this.loginError = false;
                    setTimeout(() => { lucide.createIcons(); }, 100);
                },
                logout() {
                    localStorage.removeItem('wedding_auth');
                    this.isLoggedIn = false;
                    this.emailInput = '';
                },
                addBudget() {
                    let cat = prompt('Nome da Categoria:');
                    let val = prompt('Valor (€):');
                    if(cat) { this.budget.push({ cat, val: val || '0', comp: val || '0' }); this.saveData(); }
                },
                addSupplier() {
                    let nome = prompt('Nome do Fornecedor:');
                    let servico = prompt('Serviço:');
                    let valor = prompt('Valor (€):');
                    if(nome) { this.suppliers.push({ nome, servico: servico || 'Geral', contacto: '910000000', valor: valor || '0', estado: 'Contratado' }); this.saveData(); }
                },
                addPayment() {
                    let desc = prompt('Descrição:');
                    let val = prompt('Valor (€):');
                    if(desc) { this.pagamentos.push({ descricao: desc, valor: val || '0', data: '2027-06-01', estado: 'Agendado' }); this.saveData(); }
                },
                addQuinta() {
                    let nome = prompt('Nome da Quinta:');
                    let local = prompt('Localidade:');
                    let total = prompt('Total (€):');
                    if(nome) { this.quintas.push({ nome, local: local || 'Portugal', total: total || '15000', nota: '4.5' }); this.saveData(); }
                },
                addGuest() {
                    let nome = prompt('Nome do Convidado:');
                    let grupo = prompt('Grupo:');
                    if(nome) { this.guests.push({ nome, grupo: grupo || 'Geral', rsvp: 'Pendente', prato: 'Carne' }); this.saveData(); }
                },
                toggleRsvp(index) {
                    const states = ['Confirmado', 'Pendente', 'Recusado'];
                    let current = states.indexOf(this.guests[index].rsvp);
                    this.guests[index].rsvp = states[(current + 1) % states.length];
                    this.saveData();
                },
                addMesa() {
                    let nome = prompt('Identificação da Mesa:');
                    if(nome) { this.mesas.push({ nome, lugares: '10', estado: 'Vazia' }); this.saveData(); }
                },
                addTask() {
                    let tarefa = prompt('Descrição da Tarefa:');
                    let resp = prompt('Responsável:');
                    if(tarefa) { this.tasks.push({ tarefa, resp: resp || 'Ambos', estado: 'Por iniciar' }); this.saveData(); }
                },
                toggleTask(index) {
                    this.tasks[index].estado = this.tasks[index].estado === 'Concluída' ? 'Por iniciar' : 'Concluída';
                    this.saveData();
                },
                addHoneymoonItem() {
                    let rubrica = prompt('Rubrica (Ex: Passeios):');
                    let estimado = prompt('Valor Estimado (€):');
                    if(rubrica) { this.honeymoon.push({ rubrica, estimado: estimado || '0', real: '0' }); this.saveData(); }
                },
                addGift() {
                    let convidado = prompt('Nome do Convidado:');
                    let presente = prompt('Presente:');
                    let valor = prompt('Valor (€):');
                    if(convidado) { this.gifts.push({ convidado, presente: presente || 'Outro', valor: valor || '0', agradecido: 'Não' }); this.saveData(); }
                },
                addInspiration() {
                    let categoria = prompt('Categoria (Ex: Decoração):');
                    let referencia = prompt('Referência:');
                    if(categoria) { this.inspirations.push({ categoria, referencia: referencia || 'Detalhes', prioridade: 'Alta' }); this.saveData(); }
                }
            }))
        });
