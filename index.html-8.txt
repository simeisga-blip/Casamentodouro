                <div class="space-y-3">
                    <template x-for="(insp, i) in inspirations" :key="i">
                        <div class="bg-white rounded-2xl shadow-sm border border-slate-100 overflow-hidden text-xs">
                            <img :src="getSmartImage(insp.categoria, insp.referencia)" alt="Inspiração" class="w-full h-36 object-cover">
                            <div class="p-4 flex justify-between items-center">
                                <div>
                                    <p class="font-bold text-slate-900 text-sm" x-text="insp.categoria + ': ' + insp.referencia"></p>
                                    <p class="text-slate-500" x-text="'Prioridade: ' + insp.prioridade"></p>
                                </div>
                                <button @click="inspirations.splice(i,1); saveData()" class="text-rose-500 font-bold">Eliminar</button>
                            </div>
                        </div>
                    </template>
                </div>
            </div>

        </main>

    </div>

    <!-- Script Application Logic -->
    <script>
        document.addEventListener('alpine:init', () => {
            Alpine.data('weddingApp', () => ({
                isLoggedIn: localStorage.getItem('wedding_auth') === 'true',
                emailInput: '',
                loginError: false,
                errorMessage: '',
                activeTab: 'dashboard',
                
                tabs: [
                    { id: 'dashboard', name: 'Dashboard', emoji: '📊' },
                    { id: 'config', name: 'Configuração', emoji: '⚙️' },
                    { id: 'orcamento', name: 'Orçamento', emoji: '💰' },
                    { id: 'fornecedores', name: 'Fornecedores', emoji: '🤝' },
                    { id: 'pagamentos', name: 'Pagamentos', emoji: '💳' },
                    { id: 'quintas', name: 'Quintas', emoji: '🏰' },
                    { id: 'convidados', name: 'Convidados', emoji: '👥' },
                    { id: 'mesas', name: 'Mesas e Menu', emoji: '🍽️' },
                    { id: 'agenda', name: 'Agenda', emoji: '📋' },
                    { id: 'luademel', name: 'Lua de Mel', emoji: '✈️' },
                    { id: 'presentes', name: 'Presentes', emoji: '🎁' },
                    { id: 'inspiracao', name: 'Inspiração', emoji: '✨' }
                ],
                
                config: JSON.parse(localStorage.getItem('w_config')) || { noiva: 'Inês', noivo: 'Tiago', data: '22/06/2027', local: 'Quinta das Oliveiras, Braga' },
                budget: JSON.parse(localStorage.getItem('w_budget')) || [],
                suppliers: JSON.parse(localStorage.getItem('w_suppliers')) || [],
                pagamentos: JSON.parse(localStorage.getItem('w_pagamentos')) || [],
                quintas: JSON.parse(localStorage.getItem('w_quintas')) || [
                    { nome: 'Quinta Vale Verde', local: 'Braga, Portugal', total: '15607', nota: '4.7' }
                ],
                guests: JSON.parse(localStorage.getItem('w_guests')) || [],
                mesas: JSON.parse(localStorage.getItem('w_mesas')) || [
                    { nome: 'Mesa Principal', lugares: 6, assentos: {} }
                ],
                tasks: JSON.parse(localStorage.getItem('w_tasks')) || [],
                honeymoon: JSON.parse(localStorage.getItem('w_honeymoon')) || [
                    { rubrica: 'Resort Maldivas', local: 'Atol de Ari Sul, Maldivas', estimado: '3500', real: '2100' }
                ],
                gifts: JSON.parse(localStorage.getItem('w_gifts')) || [],
                inspirations: JSON.parse(localStorage.getItem('w_inspirations')) || [
                    { categoria: 'Vestido', referencia: 'Vermelho com amarelo', prioridade: 'Alta' }
                ],

                get totalBudget() {
                    return this.budget.reduce((acc, curr) => acc + Number(curr.val || 0), 0).toLocaleString();
                },
                get confirmedGuestsCount() {
                    return this.guests.filter(g => g.rsvp === 'Confirmado').length;
                },

                getDishEmoji(prato) {
                    let p = (prato || '').toLowerCase();
                    if(p.includes('carne')) return '🥩';
                    if(p.includes('peixe')) return '🐟';
                    if(p.includes('vegetariano') || p.includes('veggie')) return '🥗';
                    if(p.includes('infantil') || p.includes('criança')) return '👶';
                    return '🍽️';
                },

                getGuestDishEmoji(guestName) {
                    if(!guestName) return '';
                    let found = this.guests.find(g => g.nome.toLowerCase() === guestName.toLowerCase());
                    if(found) {
                        return this.getDishEmoji(found.prato);
                    }
                    return '👤';
                },

                getSmartImage(cat, ref) {
                    let text = (cat + ' ' + ref).toLowerCase();
                    if(text.includes('vestido') || text.includes('noiva')) {
                        return 'https://images.unsplash.com/photo-1594552072238-b8a3272be66f?w=500&auto=format&fit=crop&q=60';
                    }
                    if(text.includes('vermelho') || text.includes('amarelo') || text.includes('cor')) {
                        return 'https://images.unsplash.com/photo-1515886657613-9f3515b0c78f?w=500&auto=format&fit=crop&q=60';
                    }
                    if(text.includes('aliança') || text.includes('anel') || text.includes('joia')) {
                        return 'https://images.unsplash.com/photo-1605100804763-247f67b3557e?w=500&auto=format&fit=crop&q=60';
                    }
                    if(text.includes('flor') || text.includes('bouquet') || text.includes('decoração')) {
                        return 'https://images.unsplash.com/photo-1519741497674-611481863552?w=500&auto=format&fit=crop&q=60';
                    }
                    if(text.includes('bolo') || text.includes('doce')) {
                        return 'https://images.unsplash.com/photo-1535141192574-5d4897c13136?w=500&auto=format&fit=crop&q=60';
                    }
                    return 'https://images.unsplash.com/photo-1511285560929-80b456fea0bc?w=500&auto=format&fit=crop&q=60';
                },

                saveData() {
                    localStorage.setItem('w_config', JSON.stringify(this.config));
                    localStorage.setItem('w_budget', JSON.stringify(this.budget));
                    localStorage.setItem('w_suppliers', JSON.stringify(this.suppliers));
                    localStorage.setItem('w_pagamentos', JSON.stringify(this.pagamentos));
                    localStorage.setItem('w_quintas', JSON.stringify(this.quintas));
                    localStorage.setItem('w_guests', JSON.stringify(this.guests));
                    localStorage.setItem('w_mesas', JSON.stringify(this.mesas));
                    localStorage.setItem('w_tasks', JSON.stringify(this.tasks));
                    localStorage.setItem('w_honeymoon', JSON.stringify(this.honeymoon));
                    localStorage.setItem('w_gifts', JSON.stringify(this.gifts));
                    localStorage.setItem('w_inspirations', JSON.stringify(this.inspirations));
                },

                handleLogin() {
                    if(!this.emailInput || !this.emailInput.includes('@')) {
                        this.loginError = true;
                        this.errorMessage = 'Insira um e-mail válido.';
                        return;
                    }
                    localStorage.setItem('wedding_auth', 'true');
                    this.isLoggedIn = true;
                    this.loginError = false;
                    setTimeout(() => { lucide.createIcons(); }, 100);
                },
                logout() {
                    localStorage.removeItem('wedding_auth');
                    this.isLoggedIn = false;
                    this.emailInput = '';
                },
                addBudget() {
                    let cat = prompt('Nome da Categoria:');
                    let val = prompt('Valor Planeado (€):');
                    if(cat) { this.budget.push({ cat, val: val || '0' }); this.saveData(); }
                },
                addSupplier() {
                    let nome = prompt('Nome do Fornecedor:');
                    let servico = prompt('Serviço prestado:');
                    let contacto = prompt('Contacto (Telefone):');
                    let valor = prompt('Valor (€):');
                    if(nome) { this.suppliers.push({ nome, servico: servico || 'Geral', contacto: contacto || 'N/A', valor: valor || '0', estado: 'Contratado' }); this.saveData(); }
                },
                addPayment() {
                    let desc = prompt('Descrição do Pagamento:');
                    let val = prompt('Valor (€):');
                    let data = prompt('Data Prevista (DD/MM/AAAA):');
                    let estado = prompt('Estado (Agendado / Pago):');
                    if(desc) { this.pagamentos.push({ descricao: desc, valor: val || '0', data: data || '01/06/2027', estado: estado || 'Agendado' }); this.saveData(); }
                },
                addQuinta() {
                    let nome = prompt('Nome da Quinta:');
                    let local = prompt('Endereço Exato ou Cidade (Ex: Av. dos Aliados, Porto):');
                    let total = prompt('Total (€):');
                    let nota = prompt('Nota (Ex: 4.8):');
                    if(nome) { this.quintas.push({ nome, local: local || 'Endereço não informado', total: total || '15000', nota: nota || '4.5' }); this.saveData(); }
                },
                addGuest() {
                    let nome = prompt('Nome do Convidado:');
                    let grupo = prompt('Grupo (Ex: Família):');
                    let prato = prompt('Prato (Carne, Peixe, Vegetariano, Infantil):');
                    if(nome) { this.guests.push({ nome, grupo: grupo || 'Geral', rsvp: 'Pendente', prato: prato || 'Carne' }); this.saveData(); }
                },
                toggleRsvp(index) {
                    const states = ['Confirmado', 'Pendente', 'Recusado'];
                    let current = states.indexOf(this.guests[index].rsvp);
                    this.guests[index].rsvp = states[(current + 1) % states.length];
                    this.saveData();
                },
                addMesa() {
                    let nome = prompt('Nome da Mesa:');
                    let lugares = prompt('Número de Lugares (Ex: 4, 6, 8):');
                    if(nome) { this.mesas.push({ nome, lugares: Number(lugares) || 6, assentos: {} }); this.saveData(); }
                },
                assignSeat(mesaIndex, seatNum) {
                    let nomeConvidado = prompt(`Nome do convidado para o Lugar ${seatNum}:`);
                    if(nomeConvidado !== null) {
                        if(nomeConvidado.trim() === '') {
                            delete this.mesas[mesaIndex].assentos[seatNum];
                        } else {
                            this.mesas[mesaIndex].assentos[seatNum] = nomeConvidado;
                        }
                        this.saveData();
                    }
                },
                addTask() {
                    let tarefa = prompt('Descrição da Tarefa:');
                    let resp = prompt('Responsável:');
                    if(tarefa) { this.tasks.push({ tarefa, resp: resp || 'Ambos', estado: 'Por iniciar' }); this.saveData(); }
                },
                toggleTask(index) {
                    this.tasks[index].estado = this.tasks[index].estado === 'Concluída' ? 'Por iniciar' : 'Concluída';
                    this.saveData();
                },
                addHoneymoonItem() {
                    let rubrica = prompt('Destino ou Despesa:');
                    let local = prompt('Endereço ou Local Exato (Ex: Ilhas Maurício):');
                    let estimado = prompt('Valor Estimado (€):');
                    let real = prompt('Gasto Real (€):');
                    if(rubrica) { this.honeymoon.push({ rubrica, local: local || 'Local a definir', estimado: estimado || '0', real: real || '0' }); this.saveData(); }
                },
                addGift() {
                    let convidado = prompt('Convidado:');
                    let presente = prompt('Presente:');
                    let valor = prompt('Valor (€):');
                    if(convidado) { this.gifts.push({ convidado, presente: presente || 'Outro', valor: valor || '0', agradecido: 'Não' }); this.saveData(); }
                },
                addInspiration() {
                    let categoria = prompt('Categoria (Ex: Vestido, Decoração):');
                    let referencia = prompt('Detalhes / Referência (Ex: Vestido amarelo e preto):');
                    if(categoria) { this.inspirations.push({ categoria, referencia, prioridade: 'Alta' }); this.saveData(); }
                }
            }))
        });

        document.addEventListener('DOMContentLoaded', () => {
            setTimeout(() => { lucide.createIcons(); }, 100);
        });
        document.addEventListener('alpine:updated', () => {
            lucide.createIcons();
        });
    </script>
